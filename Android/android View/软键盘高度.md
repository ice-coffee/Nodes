//记录原始窗口高度
private int mWindowHeight = 0;
//软键盘是否打开
private boolean isOpen;

private ViewTreeObserver.OnGlobalLayoutListener mGlobalLayoutListener = () -> {
    Window window = getWindow();
    if (null != window) {
        Rect r = new Rect();
        //获取当前窗口实际的可见区域
        window.getDecorView().getWindowVisibleDisplayFrame(r);
        int height = r.height();

        if (mWindowHeight == 0) {
            //一般情况下，这是完整的窗口高度
            mWindowHeight = height;
        } else {
            int softKeyboardHeight;
            if (mWindowHeight != height) {
                //两次窗口高度相减，就是软键盘高度
                softKeyboardHeight = mWindowHeight - height;
            } else {
                softKeyboardHeight = 0;
            }
            RelativeLayout.LayoutParams params = (RelativeLayout.LayoutParams) bottomLayout.getLayoutParams();
            if (isOpen) {
                if (softKeyboardHeight <= 0) {
                    isOpen = false;
                    params.bottomMargin = softKeyboardHeight;
                }
            } else {
                //部分手机(IQOO)会出现软键盘高度跳动的问题，原因不明。
                if (params.bottomMargin < softKeyboardHeight) {
                    isOpen = true;
                    params.bottomMargin = softKeyboardHeight;
                }
            }
            bottomLayout.setLayoutParams(params);
        }
    }
};
